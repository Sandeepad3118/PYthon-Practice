Diagonal Traversal

    print(matrix[i][i])  # 1, 5, 9

# Anti-diagonal (top-right to bottom-left)
n = len(matrix)
for i in range(n):
    print(matrix[i][n-1-i])  # 3, 5, 7

Spiral Traversal


def spiral_order(matrix):
    result = []
    while matrix:
        # Right
        result += matrix.pop(0)
        # Down
        if matrix and matrix[0]:
            for row in matrix:
                result.append(row.pop())
        # Left
        if matrix:
            result += matrix.pop()[::-1]
        # Up
        if matrix and matrix[0]:
            for row in matrix[::-1]:
                result.append(row.pop(0))
    return result

6. Neighbor Access Patterns
4-direction neighbors (Up, Down, Left, Right)


def get_neighbors(matrix, i, j):
    neighbors = []
    directions = [(-1,0), (1,0), (0,-1), (0,1)]  # U,D,L,R
    
    for dx, dy in directions:
        ni, nj = i + dx, j + dy
        if 0 <= ni < len(matrix) and 0 <= nj < len(matrix[0]):
            neighbors.append((ni, nj))
    return neighbors

8-direction neighbors (Includes diagonals)


def get_all_neighbors(matrix, i, j):
    neighbors = []
    for dx in [-1, 0, 1]:
        for dy in [-1, 0, 1]:
            if dx == 0 and dy == 0:  # Skip the cell itself
                continue
            ni, nj = i + dx, j + dy
            if 0 <= ni < len(matrix) and 0 <= nj < len(matrix[0]):
                neighbors.append((ni, nj))
    return neighbors

def search_matrix(matrix, target):
    for i in range(len(matrix)):
        for j in range(len(matrix[0])):
            if matrix[i][j] == target:
                return True

Number of Islands


def num_islands(grid):
    if not grid:
        return 0
    
    def dfs(i, j):
        if i < 0 or i >= len(grid) or j < 0 or j >= len(grid[0]) or grid[i][j] != '1':
            return
        grid[i][j] = '0'  # Mark as visited
        # Visit all neighbors
        dfs(i+1, j); dfs(i-1, j); dfs(i, j+1); dfs(i, j-1)
    
    count = 0
    for i in range(len(grid)):
        for j in range(len(grid[0])):
            if grid[i][j] == '1':
                count += 1
                dfs(i, j)
    return count

8. Performance Tips

    Time Complexity: O(m × n) for m × n matrix

    Space Complexity: O(1) for simple traversals, O(m × n) for operations requiring storage

    Use enumerate() when you need indices

    Precompute bounds to avoid repeated len() calls

    Use list comprehensions for simple transformations

Key Points:

    Row-major is most common (left→right, top→bottom)

    Always check boundaries when accessing neighbors

    Consider the problem requirements when choosing traversal pattern

    Practice common patterns like spiral, diagonal, and BFS/DFS traversals
